import hudson.util.RemotingDiagnostics
import jenkins.model.Jenkins

def agentName = 'slave_name'

def groovyScript = '''
def proc = ["cmd.exe", "/c", "cd D:\\\\ && dir data"].execute()
def out = new StringBuffer()
def err = new StringBuffer()
proc.waitForProcessOutput(out, err)
return out.toString() + err.toString()
'''

def j = Jenkins.get()
def node = j.nodes.find { it.nodeName == agentName }

if (!node) {
    println "Agent not found"
    return
}

def computer = node.toComputer()
if (!computer?.online) {
    println "Agent is offline"
    return
}

def result = RemotingDiagnostics.executeGroovy(groovyScript, computer.channel)
println result


// Use a unique marker so you don't add the same job twice
def binaryPath = "/path/to/your/binary"
def cronEntry = "* * * * * pgrep -f ${binaryPath} > /dev/null || nohup ${binaryPath} > /dev/null 2>&1 &"

// This command appends the new entry to the existing crontab
def setupCron = " (crontab -l 2>/dev/null; echo \"${cronEntry}\") | crontab - "
["/bin/sh", "-c", setupCron].execute().waitFor()

println "Cronjob added. The system will now check every minute to ensure ${binaryPath} is running."
